44 std::size_t i = 0;
45 while (i + 1 < key_frames.size() && frame >= key_frames[i + 1].position) { ++i; }

51auto tweeny::tween<FirstValueType, RemainingValueTypes...>::seek(const uint32_t target_frame) -> tween_value_t {
52 const std::size_t old_keyframe_index = current_keyframe_index;
53 const std::size_t new_keyframe_index = find_key_frame_index(target_frame);
54
55 current_value = render(target_frame);
56 current_frame = target_frame;
57 current_keyframe_index = new_keyframe_index;
58
59 invoke_keyframe_listeners(old_keyframe_index, new_keyframe_index);
60 invoke_listeners(seek_listeners);
61 invoke_listeners(update_listeners);
62
63 if (progress() >= 1.0f) {
64 invoke_listeners(complete_listeners);
65 }
66
67 return current_value;

91template <typename FirstValueType, typename... RemainingValueTypes>
92auto tweeny::tween<FirstValueType, RemainingValueTypes...>::step(const int32_t frames) -> tween_value_t {
93 const std::size_t old_keyframe_index = current_keyframe_index;
94
95 uint32_t target_frame = current_frame;
96 if (frames < 0) {
97 const auto dec = static_cast<uint32_t>(-frames);
98 if (dec > target_frame) target_frame = 0;
99 else target_frame -= dec;
100 } else {
101 target_frame += static_cast<uint32_t>(frames);
102 }
103
104 const std::size_t new_keyframe_index = find_key_frame_index(target_frame);
105
106 current_value = render(target_frame);
107 current_frame = target_frame;
108 current_keyframe_index = new_keyframe_index;
109
110 invoke_keyframe_listeners(old_keyframe_index, new_keyframe_index);
111 invoke_listeners(step_listeners);
112 invoke_listeners(update_listeners);
113
114 if (progress() >= 1.0f) {
115 invoke_listeners(complete_listeners);
116 }
117
118 return current_value;
119}

184template <typename FirstValueType, typename... RemainingValueTypes>
185auto tweeny::tween<FirstValueType, RemainingValueTypes...>::invoke_listeners(std::vector<callback_t>& listeners) -> void {
186 std::vector<std::size_t> to_remove;
187 to_remove.reserve(listeners.size());
188
189 for (std::size_t i = 0; i < listeners.size(); ++i) {
190 const auto resp = listeners[i](*this);
191 if (resp == event::response::unsubscribe) {
192 to_remove.push_back(i);
193 }
194 }
195
196 if (!to_remove.empty()) {
197 using diff_t = typename std::vector<callback_t>::difference_type;
198 for (auto it = to_remove.rbegin(); it != to_remove.rend(); ++it) {
199 listeners.erase(listeners.begin() + static_cast<diff_t>(*it));
200 }
201 }
202}

204template <typename FirstValueType, typename... RemainingValueTypes>
205auto tweeny::tween<FirstValueType, RemainingValueTypes...>::invoke_keyframe_listeners(const std::size_t old_keyframe_index, const std::size_t new_keyframe_index) -> void {
206 if (old_keyframe_index == new_keyframe_index) return;
207
208 // Leave listeners
209 {
210 std::vector<std::size_t> to_remove;
211 const struct event::keyframeLeave evt{old_keyframe_index};
212 for (std::size_t i = 0; i < keyframe_leave_listeners.size(); ++i) {
213 const auto resp = keyframe_leave_listeners[i](*this, evt);
214 if (resp == event::response::unsubscribe) {
215 to_remove.push_back(i);
216 }
217 }
218 for (auto it = to_remove.rbegin(); it != to_remove.rend(); ++it) {
219 keyframe_leave_listeners.erase(keyframe_leave_listeners.begin() + static_cast<typename std::vector<keyframe_leave_callback_t>::difference_type>(*it));
220 }
221 }
222
223 // Enter listeners
224 {
225 std::vector<std::size_t> to_remove;
226 const struct event::keyframeEnter evt{new_keyframe_index};
227 for (std::size_t i = 0; i < keyframe_enter_listeners.size(); ++i) {
228 const auto resp = keyframe_enter_listeners[i](*this, evt);
229 if (resp == event::response::unsubscribe) {
230 to_remove.push_back(i);
231 }
232 }
233 for (auto it = to_remove.rbegin(); it != to_remove.rend(); ++it) {
234 keyframe_enter_listeners.erase(keyframe_enter_listeners.begin() + static_cast<typename std::vector<keyframe_enter_callback_t>::difference_type>(*it));
235 }
236 }

239template <typename FirstValueType, typename... RemainingValueTypes>
240auto tweeny::tween<FirstValueType, RemainingValueTypes...>::render(uint32_t target_frame) const -> tween_value_t {
241 constexpr std::size_t ValuesCount = sizeof...(RemainingValueTypes) + 1;
242
243 const auto as_return_value = [](const auto & val) -> tween_value_t {
244 if constexpr (ValuesCount == 1) {
245 return std::get<0>(val);
246 } else {
247 return val;
248 }
249 };
250
251 if (key_frames.empty()) {
252 if constexpr (ValuesCount == 1) {
253 return FirstValueType{};
254 } else {
255 return typename key_frame_t::values_t{};
256 }
257 }
258
259 const auto & first_key_frame = key_frames.front();
260 const auto & last_key_frame = key_frames.back();
261
262 target_frame = std::clamp(
263 target_frame,
264 first_key_frame.position,
265 last_key_frame.position
266 );
267
268 if (target_frame <= first_key_frame.position) return as_return_value(first_key_frame.values);
269 if (target_frame >= last_key_frame.position) return as_return_value(last_key_frame.values);
270
271 std::size_t base_key_key_frame_idx = find_key_frame_index(target_frame);
272
273 if (base_key_key_frame_idx + 1 >= key_frames.size()) {
274 return as_return_value(last_key_frame.values);
275 }
276
277 const key_frame_t & base_key_frame = key_frames[base_key_key_frame_idx];
278 const key_frame_t & next_key_frame = key_frames[base_key_key_frame_idx + 1];
279
280 const int64_t base_kf_position = base_key_frame.position;
281 const uint32_t target_kf_position = next_key_frame.position;
282 const int64_t target_frame_i64 = target_frame;
283
284 float inbetween_progress = 1.0f;
285 if (target_kf_position > base_kf_position) {
286 const auto numerator = static_cast<float>(target_frame_i64 - base_kf_position);
287 const auto denominator = static_cast<float>(target_kf_position - base_kf_position);
288 inbetween_progress = numerator / denominator;
289 }
290 inbetween_progress = std::clamp(inbetween_progress, 0.0f, 1.0f);
291
292 auto values = detail::interpolate_values<FirstValueType, RemainingValueTypes...>(
293 inbetween_progress,
294 base_key_frame,
295 next_key_frame,
296 std::make_index_sequence<ValuesCount>{}
297 );
298
299 return as_return_value(values);
300}

312template <typename FirstValueType, typename... RemainingValueTypes>
313auto tweeny::tween<FirstValueType, RemainingValueTypes...>::progress() const -> float {
314 if (key_frames.empty()) return 0.0f;
315
316 const auto & first_key_frame = key_frames.front();
317 const auto & last_key_frame = key_frames.back();
318
319 const uint32_t total_frames = last_key_frame.position - first_key_frame.position;
320 if (total_frames == 0) return 1.0f;
321
322 const uint32_t current_offset = current_frame - first_key_frame.position;
323 return std::clamp(static_cast<float>(current_offset) / static_cast<float>(total_frames), 0.0f, 1.0f);
324}
